"""
Пример своего набора признаков.

Чтобы включить — переименуйте этот файл в example_gradient.py
и запустите расчёт. В списке признаков появится Gradient.
"""
import numpy as np
from effective_features.feature_sets import register


def compute(ctx):
    """Модуль градиента яркости: насколько резко меняется яркость.

    Что доступно в контексте:
      ctx.gray          — яркость, массив (высота, ширина)
      ctx.image         — все каналы, (каналы, высота, ширина)
      ctx.band('nir')   — канал по роли из dataset.json, или None
      ctx.window_sizes  — размеры скользящих окон из настроек

    Вернуть нужно словарь: имя признака → массив той же формы, что gray.
    """
    gy, gx = np.gradient(ctx.gray)
    return {'Gradient': np.hypot(gx, gy).astype(np.float32)}


register(
    id='gradient',
    name='Модуль градиента',
    description='Резкость переходов яркости',
    compute=compute,
)
